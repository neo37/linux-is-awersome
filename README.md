# Функция-обёртка для команды ssh
```bash
# Функция-обёртка для команды ssh
ssh() {
    # Если первый аргумент равен "myhost", выводим сообщение
    if [ "$1" = "myhost" ]; then
        echo "Подключение к myhost..."
    fi
    # Вызываем оригинальную команду ssh с переданными параметрами
    command ssh "$@"
}
```
